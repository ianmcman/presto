`src/frame.rs`:
```rust
pub struct ProtoVersion { pub major: u16, pub minor: u16 }          // + Copy, Eq; impl Display as "{major}.{minor}"
pub const PROTO: ProtoVersion = ProtoVersion { major: 1, minor: 0 };
pub enum Role { Presto, Engine }                                    // Copy, Eq
pub struct Hello { pub proto: ProtoVersion, pub role: Role, pub capabilities: Vec<String>, pub engine: Option<String> }
impl Hello {
    pub fn new(role: Role, capabilities: &[&str], engine: Option<String>) -> Self; // proto = PROTO
    pub fn has(&self, cap: &str) -> bool;
    pub fn check(&self, ours: ProtoVersion) -> Result<(), ProtoError>; // Err only when major differs (D-04)
}
#[derive(Debug, thiserror::Error)]
pub enum ProtoError {
    #[error("IPC protocol major version mismatch: this side speaks {ours}, peer speaks {theirs}. Update presto or the engine so both use major version {ours_major}.")]
    MajorMismatch { ours: ProtoVersion, theirs: ProtoVersion, ours_major: u16 },
}
pub mod caps { pub const PLAYBACK: &str = "playback"; pub const QUEUE: &str = "queue"; pub const API: &str = "api"; pub const MOCK: &str = "mock"; }
#[serde(tag = "t", rename_all = "snake_case")]
pub enum Frame {
    Hello(Hello),                       // engine sends first after connect; presto replies with its own
    Cmd { id: u64, cmd: Command },      // presto -> engine; engine always answers Res with same id
    Req { id: u64, req: ApiRequest },   // presto -> engine
    Res { id: u64, outcome: Outcome },  // engine -> presto
    Evt { evt: Event },                 // engine -> presto
    Ping { seq: u64 },                  // presto -> engine
    Pong { seq: u64 },                  // engine -> presto
    Mock { id: u64, fault: FaultSpec }, // presto/test -> mock only; only if engine hello has caps::MOCK
}
```
(`ProtoError` is not a wire type, no serde derive.)

pub const ENV_SOCKET: &str = "PRESTO_SOCKET";    // engine contract (D-02): argv --socket <path> --profile <dir>, or these env vars; argv wins
pub const ENV_PROFILE: &str = "PRESTO_PROFILE";
pub type Conn = tokio_util::codec::Framed<tokio::net::UnixStream, tokio_util::codec::LinesCodec>;

#[derive(Debug, thiserror::Error)]
pub enum TransportError {
    #[error("XDG_RUNTIME_DIR is not set; pass an explicit socket path")] NoRuntimeDir,
    #[error(transparent)] Io(#[from] std::io::Error),
    #[error(transparent)] Codec(#[from] tokio_util::codec::LinesCodecError),
    #[error(transparent)] Json(#[from] serde_json::Error),
}
pub fn socket_path() -> Result<PathBuf, TransportError>;  // $XDG_RUNTIME_DIR/presto/engine.sock, NoRuntimeDir if unset
pub fn bind(path: &Path) -> Result<tokio::net::UnixListener, TransportError>;
pub async fn connect(path: &Path) -> Result<Conn, TransportError>;
pub fn framed(s: tokio::net::UnixStream) -> Conn;          // LinesCodec::new_with_max_length(MAX_LINE)
pub async fn send<S>(sink: &mut S, f: &Frame) -> Result<(), TransportError>

pub async fn recv<S>(stream: &mut S) -> Result<Option<Frame>, TransportError>   // Ok(None) on EOF
