```rust
pub trait Control: Send + Sync {
    fn send(&self, c: Command);                 // fire and forget, errors become toasts
    fn raise(&self);
    fn quit(&self);
    fn art(&self, url: &str) -> Option<std::path::PathBuf>;  // Ready path from the artwork cache only
}
#[derive(Debug, PartialEq)] pub enum Act { Cmd(Command), Raise, Quit }
#[derive(Debug, PartialEq)] pub enum Reject { NotReady, NothingPlaying }
impl Reject { pub fn message(&self) -> &'static str }  // "engine not ready" / "nothing is playing"
pub fn ready(s: &CoreState) -> bool   // engine == Ready && auth == Some(SignedIn)
pub fn plan(op: &CtlOp, s: &CoreState, pos_ms: u64) -> Result<Vec<Act>, Reject>
pub fn apply(ctl: &dyn Control, acts: Vec<Act>)   // Cmd -> ctl.send, Raise -> ctl.raise, Quit -> ctl.quit
pub struct Position(std::sync::Mutex<Clock>);
impl Position { pub fn new() -> Self; pub fn now(&self, s: &CoreState) -> u64 }  // observe(&s.player, Instant::now()) then position
```
