Contract this plan creates:
```rust
#[derive(Clone)]
pub struct DataHandle { inner: Arc<Inner> }
impl DataHandle {
  pub fn new(core: CoreHandle, paths: &Paths) -> io::Result<DataHandle>;   // call inside a tokio runtime; stores Handle::current()
  pub fn list(&self, key: ViewKey) -> watch::Receiver<ListState<Item>>;  // Library, RecentlyPlayed, Shelf
  pub fn shelves(&self) -> watch::Receiver<ListState<Shelf>>;            // Recommendations
  pub fn load_more(&self, key: &ViewKey);
  pub fn refresh(&self, key: &ViewKey);   // D-03, bypasses TTL
  pub fn retry(&self, key: &ViewKey);     // manual Retry, resets attempts
  pub fn client(&self) -> &ApiClient;
  pub fn store(&self) -> &Store;
  pub fn art(&self) -> &Arc<ArtCache>;
}
```
