pub const DEBOUNCE: Duration = Duration::from_millis(300);

#[derive(Clone, Debug, Default, PartialEq)] pub struct SearchState {
  pub term: String, pub scope: Scope, pub hints: Vec<String>, pub results: Option<SearchResults>,
  pub searching: bool, pub error: Option<UiError>, pub history: Vec<String>,
}
#[derive(Clone)] pub struct Search { /* client, store, tx: Arc<watch::Sender<SearchState>>, gen: Arc<AtomicU64>, rt: Handle */ }
impl Search {
  pub fn new(client: ApiClient, store: Store) -> Search;   // inside a runtime; loads history
  pub fn state(&self) -> watch::Receiver<SearchState>;
  pub fn input(&self, term: &str);
  pub fn submit(&self);
  pub fn set_scope(&self, scope: Scope);
  pub fn clear_history(&self);
}
pub(crate) async fn settled(gen: &AtomicU64, mine: u64, d: Duration) -> bool;

  <action>
Implement search.rs per the contract.

- `settled(gen, mine, d)`: `tokio::time::sleep(d).await; gen.load(SeqCst) == mine`.
- `input(term)`: trim; bump `gen` to `mine`; publish `term`, clear `error`. If empty: clear hints and results, reload `history` from the store (so a cache wipe shows), return. Else spawn: `if !settled(DEBOUNCE)` return; if scope Catalog fetch hints `client.catalog("search/hints", &[("term", t), ("limit", "10")])` and publish `hints` only if `gen == mine`; then `if !settled(RESULTS_PAUSE - DEBOUNCE)` return; run `full(t, mine)`.
- `submit()`: bump gen, run `full(term, mine)` immediately and `store.push_search(account, term, now_ms())` when the account key exists (in spawn_blocking), then reload `history`.
- `full(t, mine)`: set `searching: true`. Catalog: `client.catalog("search", &[("term", t), ("types", "songs,albums,artists,playlists"), ("limit", "25"), ("with", "topResults")])`. Library: `client.get(ApiRequest::get("/v1/me/library/search"))` with query `term`, `types=library-songs,library-albums,library-artists,library-playlists`, `limit=25`. Use the exact paths/params from 04-02-SUMMARY `## Shapes` if they differ, and adjust the mock route in crates/presto-engine-mock/src/catalog.rs to match. On Ok and `gen == mine`: publish `results = Some(parse_search(t, &v))`, `searching: false`. On Err and `gen == mine`: publish `error = Some(UiError { kind, attempts: 0, retry_at: None })` (no auto-retry for search; the user types or presses Enter again), `searching: false`. Stale generations are dropped silently.
- `set_scope(s)`: publish scope, clear hints/results, then behave like `submit()` without recording history if term is non-empty.
- `clear_history()`: `store.clear_search(account)`, publish empty history.
- Results are never written to the `pages` table.

Unit tests with `#[tokio::test(start_paused = true)]` for the `settled` behavior bullets.
