```rust
// mirror.rs  (derive Clone, Debug, Default, Serialize, PartialEq)
pub struct QueueMirror { pub generation: u64, pub rev: u64, pub items: Vec<QueueItem>, pub index: Option<u32> }
impl QueueMirror {
    /// Applies only if rev > self.rev. Returns whether it applied.
    pub fn apply(&mut self, rev: u64, items: Vec<QueueItem>, index: Option<u32>) -> bool;
    /// New page bridge: generation += 1, rev = 0. Items kept for display.
    pub fn new_generation(&mut self);
    pub fn ids(&self) -> Vec<String>;
}
pub struct PlayerMirror { pub state: PlayState /*default Stopped*/, pub seq: u64, pub position_ms: u64, pub duration_ms: u64,
    pub volume: f32 /*default 1.0*/, pub shuffle: bool, pub repeat: RepeatMode /*default Off*/, pub track: Option<QueueItem> }
impl PlayerMirror {
    /// PlaybackState/Progress dropped when seq < self.seq; Volume/Shuffle/Repeat/TrackChanged always applied.
    pub fn apply(&mut self, evt: &Event);
    pub fn new_generation(&mut self);   // seq = 0
}
pub struct Snapshot { pub ids: Vec<String>, pub index: u32, pub position_ms: u64, pub was_playing: bool,
    pub volume: f32, pub shuffle: bool, pub repeat: RepeatMode }
/// None when the queue is empty.
pub fn snapshot(q: &QueueMirror, p: &PlayerMirror) -> Option<Snapshot>;
/// Stable hash of the id list for the D-04 same-queue counter.
pub fn queue_key(ids: &[String]) -> u64;

// auth.rs
pub enum AuthEffect { None, ShowWindow }
#[derive(Default)] pub struct AuthMachine { pub state: Option<AuthState> }
impl AuthMachine {
    pub fn on_event(&mut self, s: AuthState) -> AuthEffect;   // SignedOut -> ShowWindow (D-06); others -> None
    pub fn on_auth_expired_outcome(&mut self);                 // SignedIn -> Expired (D-13)
    pub fn allows_traffic(&self) -> bool;                      // state == Some(SignedIn)
    pub fn blocked(&self) -> bool;                             // Some(SignedOut) | Some(Expired)
}
// state.rs additions: pub auth: Option<AuthState>, pub queue: QueueMirror, pub player: PlayerMirror
```
