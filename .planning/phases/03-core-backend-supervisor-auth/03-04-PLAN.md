```rust
// config.rs
pub struct Launch { pub program: PathBuf, pub args: Vec<OsString> }
impl Launch {
    /// <engine_dir>/node_modules/electron/dist/<path.txt>, args = [engine_dir, extra...]
    pub fn electron(engine_dir: &Path, extra: &[&str]) -> io::Result<Launch>;
}
/// Called with the 0-based attempt number before each spawn.
pub type Launcher = Arc<dyn Fn(u32) -> Launch + Send + Sync>;
#[derive(Clone, Debug)]
pub struct Timings { pub heartbeat: Duration, pub misses: u32, pub connect: Duration, pub drift: Duration,
    pub backoff_base: Duration, pub backoff_cap: Duration, pub fast: Duration, pub stable: Duration,
    pub kill_grace: Duration, pub quit_wait: Duration }
// Default: HEARTBEAT_INTERVAL, HEARTBEAT_MISSES, 30s, 15s, 1s, 30s, 60s, 120s, 3s, 10s
pub const REQUIRED_BRIDGE_CAPS: &[&str] = &["playback", "queue", "api"];
pub struct CoreConfig { pub launcher: Launcher, pub socket: PathBuf, pub paths: Paths, pub timings: Timings }
/// Pure D-16 check. Err(reason) when musickit_build is None or a required cap is absent.
pub fn check_bridge(caps: &[String], musickit_build: Option<&str>) -> Result<(), String>;

// state.rs  (derive Clone, Debug, Default, Serialize, PartialEq where possible)
pub enum EngineStatus { #[default] Starting, Ready, Restarting { attempt: u32, delay_ms: u64 },
    Drift { reason: String, log_path: PathBuf }, Failed { reason: String, log_path: PathBuf, log_tail: Vec<String> } }
pub struct BridgeInfo { pub version: String, pub capabilities: Vec<String>, pub musickit_build: Option<String> }
pub struct CoreState { pub engine: EngineStatus, pub bridge: Option<BridgeInfo>, pub restarts: u32, pub log_path: Option<PathBuf> }

// lib.rs
pub struct Core;
impl Core { pub async fn start(cfg: CoreConfig) -> io::Result<CoreHandle>; }
#[derive(Clone)] pub struct CoreHandle { /* mpsc::UnboundedSender<Msg>, watch::Receiver<CoreState> */ }
impl CoreHandle {
    pub fn state(&self) -> watch::Receiver<CoreState>;
    pub async fn command(&self, cmd: Command) -> Outcome;
    pub async fn request(&self, req: ApiRequest) -> Outcome;
    pub async fn show_sign_in(&self) -> Outcome;        // Command::ShowWindow{show:true}
    pub async fn mock(&self, fault: FaultSpec) -> Outcome; // Frame::Mock, test/demo control
    pub fn restart_engine(&self);
    pub async fn shutdown(self);
}
```

    6. Failure: fail every pending and queued sender with Unavailable "engine restarted" (never replay user commands), kill the group (`killpg(pgid, SIGTERM)`, wait up to `kill_grace` for `child.wait()`, then `killpg(pgid, SIGKILL)` and wait; ignore ESRCH), remove the pidfile, `state.restarts += 1`, then `backoff.on_failure(now)`: Some(d) -> status `Restarting{attempt: n+1, delay_ms}`, sleep d while still serving messages (Cmd/Req fail Unavailable; Restart skips the wait and calls `backoff.reset()`; Shutdown exits); None -> status `Failed{reason, log_path, log_tail: tail(&log_path, 40)}` and wait for Restart (reset backoff, new attempt) or Shutdown.

    8.
